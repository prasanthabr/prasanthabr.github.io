{{/* badge.html — one taxonomy term, rendered as a pill. Params: taxonomy (string, required) plural taxonomy name, e.g. "statuses" term (string, required) the raw term as written in front matter link (bool, default true) render as to the term page Colour comes from `--hue--` when `data/taxonomy.toml` gives the term a hue, otherwise `--hue-default`. Nothing to register here. */}} {{- $taxonomy := .taxonomy }} {{- $term := .term }} {{- $slug := $term | urlize }} {{- $link := true }}{{ if isset . "link" }}{{ $link = .link }}{{ end }} {{- $terms := partialCached "term-config.html" . }} {{- $help := (index $terms (printf "%s/%s" $taxonomy $slug) | default dict).help }} {{- $class := cond (eq $taxonomy "tags") "badge badge--tag" "badge" }} {{- $style := printf "--hue:var(--hue-%s-%s, var(--hue-default))" $taxonomy $slug }} {{- if $link -}} {{ $term }} {{- else -}} {{ $term }} {{- end -}}
