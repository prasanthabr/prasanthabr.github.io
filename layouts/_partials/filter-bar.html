{{/* filter-bar.html — the slicing controls above an index. Params: pages (page collection, required) the notes the bar is filtering Counts are computed from `pages`, not site-wide, so they always match what is actually on screen. Groups and their order come from `filter-taxonomies.html`; term order comes from the `order` list in `data/taxonomy.toml`, with anything unlisted falling to the end by count. Rendered `hidden`; `js/filter.js` reveals it. Without JavaScript the reader gets the full unfiltered list rather than dead controls. */}} {{- $pages := .pages }} {{- $facets := partial "filter-taxonomies.html" . }}
{{- range $facets }} {{- $taxonomy := . }} {{- $cfg := index hugo.Data.taxonomy $taxonomy | default dict }} {{/* Count each term across the pages on this index. */}} {{- $counts := dict }} {{- range $pages }} {{- range .GetTerms $taxonomy }} {{- $t := .Title }} {{- $counts = merge $counts (dict $t (add (index $counts $t | default 0) 1)) }} {{- end }} {{- end }} {{- if $counts }} {{/* Explicitly ordered terms first, then the rest by count descending. Matching is on slugs because $counts is keyed by display title. */}} {{- $ordered := slice }} {{- $taken := slice }} {{- range $cfg.order }} {{- $want := . | urlize }} {{- range $term, $n := $counts }} {{- if eq ($term | urlize) $want }} {{- $ordered = $ordered | append $term }}{{ $taken = $taken | append $term }} {{- end }} {{- end }} {{- end }} {{- $rest := slice }} {{- range $term, $n := $counts }} {{- if not (in $taken $term) }} {{- $rest = $rest | append (dict "term" $term "n" $n) }} {{- end }} {{- end }} {{- range sort $rest "n" "desc" }}{{ $ordered = $ordered | append .term }}{{ end }}
{{ $cfg.label | default (humanize $taxonomy) }} {{- $termCfg := partialCached "term-config.html" . }} {{- range $ordered }} {{- $help := (index $termCfg (printf "%s/%s" $taxonomy (. | urlize)) | default dict).help }} {{- end }}
{{- end }} {{- end }}
